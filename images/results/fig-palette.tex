% Colours and axis style of the Results figures, defined once. Load it in the preamble, after pgfplots
% and the thesis TikZ libraries. The arm colours are the full-strength families of the preamble's
% farmRM (cyan) and farmLLM (violet); those two are tints for box fills and too pale for bars.
% One scheme in every figure: solid = learned or built at the first round, light = built after a further
% round, open = built but not learned, hatched (the arm's colour) = not built.
\usetikzlibrary{patterns}
\colorlet{plotFarm}{cyan!65!black}          % FARM
\colorlet{plotArm}{violet!85!white}         % ARM-FM
\colorlet{plotNoRm}{black!50}               % training with no Reward Machine
\colorlet{plotFarmLight}{plotFarm!42!white}
\colorlet{plotArmLight}{plotArm!42!white}
\colorlet{plotNoRmLight}{plotNoRm!42!white}
\colorlet{plotPos}{green!55!black}          % machine arrows: pays
\colorlet{plotNeg}{red!75!black}            % machine arrows: only costs
\colorlet{plotZero}{black!50}               % machine arrows: pays nothing, or both signs
\pgfplotsset{res axis/.style={
    axis line style={black!60}, tick style={black!60},
    tick label style={font=\footnotesize}, label style={font=\footnotesize},
    legend cell align=left,
    legend style={font=\footnotesize, draw=none, fill=none, /tikz/every even column/.append style={column sep=8pt}},
}}
