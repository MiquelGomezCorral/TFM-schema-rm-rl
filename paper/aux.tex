\begin{algorithm}
\caption{Supervisor step}
\begin{algorithmic}[1]
\Require MARL environment $E$, meta-state $(o^i, L_i)$, current environment state $s_t$, meta-action $(\text{assign } a)$
\Ensure Next meta-state $(o^{i+1}, L_{i+1})$, meta-reward $r$
\Procedure{Step}{$E, (o^i, L_i), s_t, (\text{assign } a)$}
    \If{$L_i.\text{count}(-) > 1$}
        \State $L_{i+1} \gets L_i$
        \State $k \gets L_i.\text{getFirstPosition}(-)$
        \State $L_{i+1}[k] \gets a$
        \State $o^{i+1} \gets \mathcal{F}_{i+1}(s_t)$
        \State $r \gets 0$
    \Else
        \State $k \gets L_i.\text{getLastPosition}()$
        \State $L_i[k] \gets a$
        \State $(s_{t+1}, R) \gets E.\text{execute}(s_t, L_i)$
        \State $\mathbf{r} \gets R.\text{aggregate}()$
        \State $L_{i+1} \gets (-, \ldots, -)$
        \State $o^{i+1} \gets \mathcal{F}_{1}(s_{t+1})$
    \EndIf
    \State \Return $(o^{i+1}, L_{i+1}), r$
\EndProcedure
\end{algorithmic}
\end{algorithm}
